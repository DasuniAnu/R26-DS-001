46def _fuse_youtube_hate_predictions(audio_prediction: dict, transcript_prediction: dict | None,
47                                 asr_needs_review=False) -> dict:
48    """Fuse audio and transcript hate predictions for a YouTube segment only."""
49    audio_label = audio_prediction["hate_label"]
50    text_label = transcript_prediction["prediction"] if transcript_prediction else "NOT_ASSESSED"
51
52    if audio_label == "NO_SPEECH":
53        label = "NOT_ASSESSED"
54        is_offensive = False
55        reason = "No speech was detected, so transcript hate analysis was not assessed."
56    elif audio_label == "OFF" or text_label == "OFF":
57        label = "OFF"
58        is_offensive = True
59        sources = []
60        if audio_label == "OFF":
61            sources.append("audio model")
62        if text_label == "OFF":
63            sources.append("Sinhala ASR transcript")
64        reason = "Offensive content flagged by " + " and ".join(sources) + "."
65    elif asr_needs_review or (audio_label == "UNCERTAIN" and transcript_prediction is None):
66        label = "UNCERTAIN"
67        is_offensive = False
68        reason = "A reliable combined result is unavailable; the audio or transcript needs review."
69    else:
70        label = "NOT"
71        is_offensive = False
72        reason = "Neither the audio model nor the transcript model flagged offensive content."
73
74    return {
75        "hate_speech": is_offensive,
76        "hate_label": label,
77        "audio_hate_label": audio_label,
78        "transcript_hate_label": text_label,
79        "fusion_reason": reason,
80    }

83def _is_usable_sinhala_transcript(text: str) -> bool:
84    """Reject decoder hallucinations before they reach the text hate model."""
85    meaningful = [char for char in text if not char.isspace() and char.isalnum()]
86    if not meaningful:
87        return False
88    sinhala_chars = sum("\u0D80" <= char <= "\u0DFF" for char in meaningful)
89    return sinhala_chars >= 3 and sinhala_chars / len(meaningful) >= 0.25

139        "zero_crossing_rate": round(zcr, 4),

116def analyze_youtube_video(url: str):
117    metadata = get_video_metadata(url)
118    duration = metadata["duration"]
119    if duration <= 0:
120        raise ValueError("Could not determine video duration or video has no audio.")
121    if duration > MAX_VIDEO_DURATION:
122        raise ValueError(
123            f"Video too long ({format_time(duration)}). "
124            f"Max allowed: {format_time(MAX_VIDEO_DURATION)} (10 min)."
125        )
126
127    filepath, _ = download_youtube_audio(url)
128    if not filepath or not os.path.exists(filepath):
129        raise RuntimeError("Audio extraction failed. Verify FFmpeg is accessible.")
130
131    try:
132        try:
133            audio, _ = librosa.load(filepath, sr=SAMPLE_RATE, mono=True)
134        except Exception as e:
135            raise RuntimeError(f"Failed to load audio. Verify FFmpeg and audio codecs. Details: {str(e)}")
136
137        total_samples = len(audio)
138        if total_samples == 0:
139            raise RuntimeError("Audio file is empty.")
140
141        # Keep the original waveform for ASR. VAD masking below is useful for
142        # audio classification, but it can remove phonetic detail needed by
143        # the speech recognizer to produce an accurate transcript.
144        asr_audio = audio.copy()
145
146        # --- VAD MASKING: Keep only speech, zero out music/noise ---
147        try:
148            model, utils = _load_vad()
149            get_speech_timestamps = utils[0]
150            audio_tensor = torch.from_numpy(audio)
151            
152            # Get timestamps of speech
153            speech_timestamps = get_speech_timestamps(audio_tensor, model, sampling_rate=SAMPLE_RATE)
154            
155            # Create a boolean mask and apply it
156            mask = np.zeros(len(audio), dtype=bool)
157            for ts in speech_timestamps:
158                mask[ts['start']:ts['end']] = True
159                
160            audio = audio * mask
161        except Exception as e:
162            print(f"[VAD Warning] Failed to apply VAD mask, falling back to raw audio. Details: {e}")
163        # -----------------------------------------------------------
164
165        segments = []
166        for start in range(0, total_samples, SEGMENT_SAMPLES):
167            end = min(start + SEGMENT_SAMPLES, total_samples)
168            seg = audio[start:end].copy()
169            asr_seg = asr_audio[start:end].copy()
170            context_start = max(0, start - ASR_CONTEXT_SAMPLES)
171            context_end = min(total_samples, end + ASR_CONTEXT_SAMPLES)
172
173            if len(seg) < SEGMENT_SAMPLES:
174                seg = np.pad(seg, (0, SEGMENT_SAMPLES - len(seg)), mode="constant")
175            if len(asr_seg) < SEGMENT_SAMPLES:
176                asr_seg = np.pad(asr_seg, (0, SEGMENT_SAMPLES - len(asr_seg)), mode="constant")
177
178            peak = np.max(np.abs(seg))
179            if peak > 0:
180                seg = seg / peak
181
182            start_t = start / SAMPLE_RATE
183            end_t   = min(end / SAMPLE_RATE, float(duration))
184            segments.append({
185                "start_time": round(start_t, 2),
186                "end_time":   round(end_t, 2),
187                "start_fmt":  format_time(start_t),
188                "end_fmt":    format_time(end_t),
189                "audio":      seg.astype(np.float32),
190                "asr_audio":  asr_seg.astype(np.float32),
191                "asr_context": asr_audio[context_start:context_end].copy(),
192                "asr_core_start": start - context_start,
193                "asr_core_end": end - context_start,
194            })

45def decode_segment(logits, processor, blank_id, frame_start=0, frame_end=None):
46    """Collapse CTC runs before selecting the central interval.
47
48    Assign each run by its midpoint, preserving boundary context without
49    including entire neighbouring transcripts. Blank frames never inflate scores.
50    """
51    probabilities = torch.softmax(logits.float(), dim=-1)
52    scores, ids = probabilities.max(dim=-1)
53    ids, scores = ids.tolist(), scores.tolist()
54    frame_end = len(ids) if frame_end is None else frame_end
55    tokens, token_scores = [], []
56    start = 0
57    while start < len(ids):
58        end = start + 1
59        while end < len(ids) and ids[end] == ids[start]:
60            end += 1
61        midpoint = (start + end - 1) / 2
62        if ids[start] != blank_id and frame_start <= midpoint < frame_end:
63            tokens.append(ids[start])
64            token_scores.append(max(scores[start:end]))
65        start = end
66    text = processor.tokenizer.decode(
67        tokens, group_tokens=False, skip_special_tokens=True,
68    ).strip()
69    score = float(np.mean(token_scores)) if token_scores else 0.0
70    weak_fraction = (sum(s < MIN_TOKEN_SCORE for s in token_scores) / len(token_scores)
71                     if token_scores else 1.0)
72    needs_review = not text or score < MIN_TOKEN_SCORE or weak_fraction > MAX_WEAK_TOKEN_FRACTION
73    return {
74        "text": text,
75        "score": round(score, 4),
76        "needs_review": needs_review,
77        "reason": ("Recognition is uncertain; excluded from text hate classification."
78                   if needs_review else ""),
79    }

82def transcribe_sinhala_audio(audio: np.ndarray, core_start=0, core_end=None) -> dict:
83    """Transcribe original 16 kHz audio with context; return only the core interval."""
84    if audio is None or len(audio) == 0:
85        return {"text": "", "score": 0.0, "needs_review": True, "reason": "No audio samples."}
86    core_end = len(audio) if core_end is None else core_end
87    if not 0 <= core_start < core_end <= len(audio):
88        raise ValueError("Invalid transcription interval")
89    if not np.isfinite(audio).all():
90        return {"text": "", "score": 0.0, "needs_review": True, "reason": "Invalid audio samples."}
91    if np.max(np.abs(audio[core_start:core_end])) < 1e-7:
92        return {"text": "", "score": 0.0, "needs_review": True, "reason": "Silent audio interval."}
93
94    processor, model = _load_model()
95    try:
96        inputs = processor(
97            audio.astype(np.float32),
98            sampling_rate=SAMPLE_RATE,
99            return_tensors="pt",
100            padding=True,
101        )
102        with torch.inference_mode():
103            logits = model(**inputs).logits[0]
104        # Wav2Vec2 emits one frame per convolutional stride. Use its actual
105        # receptive-field centre instead of scaling by the padded clip length.
106        stride, receptive_field = 1, 1
107        for kernel, step in zip(model.config.conv_kernel, model.config.conv_stride):
108            receptive_field += (kernel - 1) * stride
109            stride *= step
110        offset = (receptive_field - 1) / 2
111        return decode_segment(
112            logits, processor, model.config.pad_token_id,
113            (core_start - offset) / stride, (core_end - offset) / stride,
114        )
115    except Exception as exc:
116        raise RuntimeError(f"Sinhala transcription failed: {exc}") from exc

1# YouTube transcription
2
3Each 10-second analysis segment is transcribed with up to 2 seconds of original
4audio on either side. The final ASR window is not padded with artificial silence.
5CTC token runs are assigned to the central segment using their frame midpoints
6and the model's convolutional stride/receptive field. Extra context text is
7excluded. These are approximate character alignments, not word-level forced
8alignment; words may still span displayed segment boundaries.
9
10Audio hate detection and deepfake detection retain their existing inputs.
11
12The recognition score is the mean of the maximum posterior in each emitted,
13non-blank CTC run. Blank frames are excluded. An initial heuristic flags a
14transcript if its mean is below 0.65 or more than 25% of its tokens score below
150.65. These constants are in `asr_manager.py`. They are NOT calibrated accuracy
16probabilities. High-scoring transcripts can still be wrong.
17
18Uncertain or unusable Sinhala transcripts are retained for review but are not
19sent to the text hate classifier. An independently offensive audio result
20remains offensive; otherwise an uncertain transcript yields an uncertain fused
21result and is excluded from the clean count. ASR errors affect only that segment.
22The existing audio model's NO_SPEECH decision still skips transcription.
23
24Run offline regression checks from the project root:
25

27
28Before claiming improved transcription accuracy, collect manually corrected
29Sinhala transcripts for clear, slow, quiet, noisy, and music-backed examples.
30Compare word/character error rates, review rate, and resulting hate predictions
31against the previous pipeline. Tune thresholds on a development split and

33and model replacement are not enabled by this change.
